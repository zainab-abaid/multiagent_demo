-- id 200: How many customers are there in total?
SELECT COUNT(*) FROM Customer;

-- id 201: How many employees work as 'Sales Support Agent'?
SELECT COUNT(*)
FROM Employee
WHERE Title = 'Sales Support Agent';

-- id 202: Which country has the highest number of customers?
SELECT Country
FROM Customer
GROUP BY Country
ORDER BY COUNT(*) DESC, Country ASC
LIMIT 1;

-- id 203: Which employee has the most customers assigned to them as support representative?
SELECT e.FirstName || ' ' || e.LastName AS EmpName
FROM Customer c
JOIN Employee e ON c.SupportRepId = e.EmployeeId
GROUP BY EmpName
ORDER BY COUNT(*) DESC, EmpName ASC
LIMIT 1;

-- id 204: What is the total revenue from invoices billed to customers in the USA?
SELECT ROUND(SUM(i.Total), 2) AS TotalRevenue
FROM Invoice i
JOIN Customer c ON i.CustomerId = c.CustomerId
WHERE c.Country = 'USA';

-- id 205: How many invoices does the customer 'Luís Gonçalves' have?
SELECT COUNT(*) AS NumInvoices
FROM Invoice
WHERE CustomerId = (
    SELECT CustomerId
    FROM Customer
    WHERE FirstName = 'Luís' AND LastName = 'Gonçalves'
);

-- id 206: What is the total revenue from customer 'František Wichterlová'?
SELECT ROUND(SUM(Total), 2) AS TotalRevenue
FROM Invoice
WHERE CustomerId = (
    SELECT CustomerId
    FROM Customer
    WHERE FirstName = 'František' AND LastName = 'Wichterlová'
);

-- id 207: Which city has the highest number of customers?
SELECT City
FROM Customer
GROUP BY City
ORDER BY COUNT(*) DESC, City ASC
LIMIT 1;

-- id 208: What is the average invoice total across all invoices?
SELECT ROUND(AVG(Total), 2) AS AvgTotal
FROM Invoice;

-- id 209: How many invoices were issued in the year 2010?
SELECT COUNT(*) AS NumInvoices2010
FROM Invoice
WHERE strftime('%Y', InvoiceDate) = '2010';

-- id 210: Which customer generated the highest total revenue across all their invoices?
SELECT c.FirstName || ' ' || c.LastName AS CustomerName
FROM Invoice i
JOIN Customer c ON i.CustomerId = c.CustomerId
GROUP BY CustomerName
ORDER BY SUM(i.Total) DESC, CustomerName ASC
LIMIT 1;

-- id 211: How many invoice line items are there in total?
SELECT COUNT(*) FROM InvoiceLine;

-- id 212: Which invoice has the highest total amount?
SELECT InvoiceId
FROM Invoice
ORDER BY Total DESC, InvoiceId ASC
LIMIT 1;

-- id 213: What is the total quantity of items sold across all invoice lines?
SELECT SUM(Quantity)
FROM InvoiceLine;

-- id 214: How many customers have never placed any invoice?
SELECT COUNT(*) AS NumWithoutInvoices
FROM Customer c
LEFT JOIN Invoice i ON c.CustomerId = i.CustomerId
WHERE i.InvoiceId IS NULL;

-- id 215: Which country generated the highest total revenue from invoices?
SELECT c.Country
FROM Invoice i
JOIN Customer c ON i.CustomerId = c.CustomerId
GROUP BY c.Country
ORDER BY SUM(i.Total) DESC, c.Country ASC
LIMIT 1;

-- id 216: What is the total revenue from all invoices issued in the year 2009?
SELECT ROUND(SUM(Total), 2) AS TotalRevenue2009
FROM Invoice
WHERE strftime('%Y', InvoiceDate) = '2009';

-- id 217: Which support agent has customers with the highest combined total spending?
SELECT e.FirstName || ' ' || e.LastName AS AgentName
FROM Customer c
JOIN Employee e ON c.SupportRepId = e.EmployeeId
JOIN Invoice i ON c.CustomerId = i.CustomerId
GROUP BY AgentName
ORDER BY SUM(i.Total) DESC, AgentName ASC
LIMIT 1;

-- id 218: How many invoices were billed to customers in Canada?
SELECT COUNT(*)
FROM Invoice i
JOIN Customer c ON i.CustomerId = c.CustomerId
WHERE c.Country = 'Canada';

-- id 219: What is the total revenue from customers in Brazil?
SELECT ROUND(SUM(i.Total), 2)
FROM Invoice i
JOIN Customer c ON i.CustomerId = c.CustomerId
WHERE c.Country = 'Brazil';
